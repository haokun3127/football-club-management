import { describe, expect, it, vi } from "vitest";
import { mkdtempSync, rmSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { createSeedData } from "../src/seed.js";
import { createPlatformPersistence } from "../src/persistence/platform-persistence.js";
import { InMemoryStore, PersistentApiStore } from "../src/store.js";
import { buildServer } from "../src/server.js";
import { HeaderMembershipResolver } from "../src/auth/context.js";
import { assessmentWorkflow } from "../src/application/assessment-workflow.js";
import { confirmedStudentResults } from "../src/application/confirmed-student-results.js";

const clubId = "club-chongqing-talent";
describe("same-course published correction ordering", () => {
  it("reads the corrected publication across a decimal ID boundary, including after reopen", async () => {
    vi.stubEnv("FCM_ASSESSMENT_PUBLICATION_CLUBS", clubId);
    // Freeze only Date: covers a same-millisecond tie without blocking async transactions.
    vi.useFakeTimers({ toFake: ["Date"] });
    vi.setSystemTime(new Date("2026-10-30T00:00:00.000Z"));
    const directory = mkdtempSync(join(tmpdir(), "fcm-record-order-"));
    const data = createSeedData();
    data.metricRecords.push({ ...data.metricRecords[0]!, id: "metric-record-9900" });
    let p = await createPlatformPersistence({ databasePath: join(directory, "api.sqlite"), seedData: data });
    let app: ReturnType<typeof buildServer> | undefined;
    try {
      let store = new PersistentApiStore(p.repositories, data);
      const task = { id: "correction-boundary", clubId, teamId: "team-u10-dev", templateId: "assessment-template-technical", termLabel: "2026", title: "Correction", startsOn: "2020-01-01", dueOn: "2099-12-31" };
      await store.saveAssessmentTask(task);
      const inputs = store.listAssessmentTestItems(clubId).filter(i => i.id.startsWith("assessment-test-cq-talent-")).map(i => ({ testItemId: i.id, eventId: "event-training-1", value: { kind: "score_0_100" as const, score: 100 } }));
      const flow = assessmentWorkflow(store);
      await flow.save(task, "student-1", "coach-1", { expectedRevision: 0, entries: inputs });
      const first = await flow.publish(task, "student-1", "coach-1", 1);
      await flow.save(task, "student-1", "coach-1", { expectedRevision: 1, reason: "核对更正", entries: [{ ...inputs[0]!, value: { kind: "score_0_100", score: 60 } }] });
      const beforePublish = store.getStudentMetrics(clubId, "student-1");
      const changedMetric = store.listAssessmentTestItems(clubId).find(i => i.id === inputs[0]!.testItemId)!.metricId;
      expect(beforePublish.find(r => r.metricId === changedMetric)?.assessmentId).toBe(first.assessmentId);
      const corrected = await flow.publish(task, "student-1", "coach-1", 2);
      const raw = await confirmedStudentResults(store, clubId, "student-1", new Set([task.teamId]));
      expect(raw.find(r => r.testItemId === inputs[0]!.testItemId)?.value).toEqual({ kind: "score_0_100", score: 60 });
      const records = store.getStudentMetrics(clubId, "student-1");
      expect(records.find(r => r.assessmentId === first.assessmentId && r.metricId === changedMetric)?.id).toBe("metric-record-9901");
      expect(records.find(r => r.assessmentId === corrected.assessmentId && r.metricId === changedMetric)?.id).toBe("metric-record-10000");
      const readGrowth = async () => {
        app = buildServer(store, { logger: false, membershipResolver: new HeaderMembershipResolver(p.repositories.users, p.repositories.memberships) });
        const growth = await app.inject({ method: "GET", url: `/clubs/${clubId}/app-clients/app-client-cq-talent-wechat-main/parent/students/student-1/growth-summary`, headers: { "x-user-id": "user-parent-1" } });
        expect(growth.statusCode).toBe(200);
        const latest = growth.json().latest.find((row: { metricId: string }) => row.metricId === changedMetric).record;
        expect(latest.value).toEqual({ kind: "score_0_100", score: 60 });
        expect(latest.assessmentId).toBe(corrected.assessmentId);
        await app.close(); app = undefined;
      };
      await readGrowth();
      p.database.close();
      p = await createPlatformPersistence({ databasePath: join(directory, "api.sqlite"), seed: false });
      store = new PersistentApiStore(p.repositories, createSeedData());
      await readGrowth();
    } finally {
      await app?.close(); p.database.close(); rmSync(directory, { recursive: true, force: true });
      vi.useRealTimers(); vi.unstubAllEnvs();
    }
  }, 15_000);

  it("keeps memory-store latest reads in occurrence, recorded-time and append order", () => {
    const data = createSeedData();
    const record = { ...data.metricRecords[0]!, occurredAt: "2026-10-01T00:00:00.000Z", createdAt: "2026-10-02T00:00:00.000Z", updatedAt: "2026-10-02T00:00:00.000Z" };
    data.metricRecords = [{ ...record, id: "metric-record-9901", value: { kind: "score_0_100", score: 100 } }, { ...record, id: "metric-record-10000", value: { kind: "score_0_100", score: 60 } }];
    const store = new InMemoryStore(data);
    expect(store.getStudentMetrics(record.clubId, record.studentId)[0]?.id).toBe("metric-record-10000");
    expect(store.listMetricRecords(record.clubId)[0]?.id).toBe("metric-record-10000");
  });
});

it('audit keeps latest chronological course when supported ISO timestamps use different offsets',async()=>{
 vi.stubEnv('FCM_ASSESSMENT_PUBLICATION_CLUBS',clubId);
 const p=await createPlatformPersistence({databasePath:':memory:'});
 const store=new PersistentApiStore(p.repositories);
 let app:ReturnType<typeof buildServer>|undefined;
 try{
  app=buildServer(store,{logger:false,membershipResolver:new HeaderMembershipResolver(p.repositories.users,p.repositories.memberships)});
  const makeCourse=async(startsAt:string,endsAt:string)=>{const response=await app!.inject({method:'POST',url:`/clubs/${clubId}/admin/calendar/events`,headers:{'x-user-id':'user-coach-1'},payload:{type:'training',title:'Audit ISO course',startsAt,endsAt,primaryTeamId:'team-u10-dev',ownerCoachId:'coach-1',status:'completed',participants:[{studentId:'student-1',status:'present'}],trainingSession:{kind:'team'}}});expect(response.statusCode).toBe(200);return response.json();};
  const older=await makeCourse('2026-10-01T10:00:00+08:00','2026-10-01T11:00:00+08:00');
  const newer=await makeCourse('2026-10-01T03:00:00.000Z','2026-10-01T04:00:00.000Z');
  expect(Date.parse(older.timeRange.startsAt)).toBeLessThan(Date.parse(newer.timeRange.startsAt));
  const item=store.listAssessmentTestItems(clubId).find(i=>i.id.startsWith('assessment-test-cq-talent-'))!;
  const task={id:'audit-iso-order',clubId,teamId:'team-u10-dev',templateId:'assessment-template-technical',testItemIds:[item.id],studentIds:['student-1'],termLabel:'2026',title:'ISO course ordering',startsOn:'2020-01-01',dueOn:'2099-12-31'};
  await store.saveAssessmentTask(task);const flow=assessmentWorkflow(store);
  await flow.save(task,'student-1','coach-1',{expectedRevision:0,entries:[{testItemId:item.id,eventId:older.id,value:{kind:'score_0_100',score:70}}]});
  await flow.publish(task,'student-1','coach-1',1);
  await flow.save(task,'student-1','coach-1',{expectedRevision:1,reason:'记录更新课程成绩',entries:[{testItemId:item.id,eventId:newer.id,value:{kind:'score_0_100',score:80}}]});
  const latest=await flow.publish(task,'student-1','coach-1',2);
  const raw=await confirmedStudentResults(store,clubId,'student-1',new Set([task.teamId]));
  expect(raw.find(r=>r.testItemId===item.id)?.value).toEqual({kind:'score_0_100',score:80});
  const response=await app!.inject({method:'GET',url:`/clubs/${clubId}/app-clients/app-client-cq-talent-wechat-main/parent/students/student-1/growth-summary`,headers:{'x-user-id':'user-parent-1'}});
  expect(response.statusCode).toBe(200);
  const record=response.json().latest.find((r:any)=>r.metricId===item.metricId).record;
  writeFileSync('/tmp/cq-round3-order-result.json',JSON.stringify({olderCourse:older.timeRange.startsAt,newerCourse:newer.timeRange.startsAt,olderUtc:new Date(older.timeRange.startsAt).toISOString(),newerUtc:new Date(newer.timeRange.startsAt).toISOString(),actual:record.value,actualAssessment:record.assessmentId,expected:{kind:'score_0_100',score:80},expectedAssessment:latest.assessmentId},null,2));
  expect(record.assessmentId).toBe(latest.assessmentId);
  expect(record.value).toEqual({kind:'score_0_100',score:80});
 }finally{await app?.close();p.database.close();vi.unstubAllEnvs();}
},15000);
