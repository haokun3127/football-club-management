import { describe, expect, it, vi } from 'vitest';
import {writeFileSync} from 'node:fs';
import { createSeedData } from '../src/seed.js';
import { createPlatformPersistence } from '../src/persistence/platform-persistence.js';
import { PersistentApiStore } from '../src/store.js';
import { buildServer } from '../src/server.js';
import { HeaderMembershipResolver } from '../src/auth/context.js';
import { assessmentWorkflow } from '../src/application/assessment-workflow.js';
import { confirmedStudentResults } from '../src/application/confirmed-student-results.js';
const clubId='club-chongqing-talent';
const coach='/clubs/'+clubId+'/app-clients/app-client-cq-talent-wechat-main/coach';
const headers={'x-user-id':'user-coach-1'};

describe('independent fixed baseline audit',()=>{
 it('observes stale classroom snapshot overwriting a newer score and partial writes before invalid-row rejection',async()=>{
  const p=await createPlatformPersistence({databasePath:':memory:'});
  const store=new PersistentApiStore(p.repositories);
  const app=buildServer(store,{logger:false,membershipResolver:new HeaderMembershipResolver(p.repositories.users,p.repositories.memberships)});
  try {
   const base=coach+'/events/event-training-1';
   const workbench=await app.inject({method:'GET',url:base+'/workbench',headers});
   expect((await app.inject({method:'PUT',url:base+'/training-projects',headers,payload:{expectedRevision:workbench.json().trainingContentRevision,projectIds:['drill-cq-talent-assessment-001','drill-cq-talent-assessment-002']}})).statusCode).toBe(200);
   expect((await app.inject({method:'PUT',url:base+'/attendance',headers,payload:{participants:[{studentId:'student-1',status:'present'}]}})).statusCode).toBe(200);
   const row=(studentId:string,score:number,trainingProjectId='drill-cq-talent-assessment-001')=>({studentId,trainingProjectId,score});
   const save=(assessments:ReturnType<typeof row>[])=>app.inject({method:'PUT',url:base+'/training-content-assessments',headers,payload:{assessments}});
   expect((await save([row('student-1',70),row('student-1',60,'drill-cq-talent-assessment-002')])).statusCode).toBe(200);
   const get=()=>app.inject({method:'GET',url:base+'/training-content-assessments',headers});
   const a=(await get()).json().assessments; const b=(await get()).json().assessments;
   // The native page submits every restored score, including unchanged cells.
   expect((await save(a.map((r:any)=>row(r.studentId,r.trainingProjectId==='drill-cq-talent-assessment-001'?90:r.score,r.trainingProjectId)))).statusCode).toBe(200);
   expect((await save(b.map((r:any)=>row(r.studentId,r.trainingProjectId==='drill-cq-talent-assessment-002'?80:r.score,r.trainingProjectId)))).statusCode).toBe(200);
   expect((await get()).json().assessments.find((r:any)=>r.trainingProjectId==='drill-cq-talent-assessment-001').score).toBe(70);
   writeFileSync('/tmp/cq-audit-concurrency-result.json',JSON.stringify((await get()).json(),null,2));
   const rejected=await save([row('student-1',50),row('not-on-roster',30)]);
   expect(rejected.statusCode).toBe(400);
   expect((await get()).json().assessments.find((r:any)=>r.studentId==='student-1').score).toBe(50);
  }finally{await app.close();p.database.close();}
 },30000);
 it('checks current correction projections across a decimal ID boundary',async()=>{
  vi.stubEnv('FCM_ASSESSMENT_PUBLICATION_CLUBS',clubId);
  const data=createSeedData();
  data.metricRecords.push({...data.metricRecords[0]!,id:'metric-record-9900'});
  const p=await createPlatformPersistence({databasePath:':memory:',seedData:data});
  const store=new PersistentApiStore(p.repositories,data);
  const app=buildServer(store,{logger:false,membershipResolver:new HeaderMembershipResolver(p.repositories.users,p.repositories.memberships)});
  try {
   const task={id:'audit-boundary',clubId,teamId:'team-u10-dev',templateId:'assessment-template-technical',termLabel:'2026',title:'Audit',startsOn:'2020-01-01',dueOn:'2099-12-31'};
   await store.saveAssessmentTask(task);
   const inputs=store.listAssessmentTestItems(clubId).filter(i=>i.id.startsWith('assessment-test-cq-talent-')).map(i=>({testItemId:i.id,eventId:'event-training-1',value:{kind:'score_0_100' as const,score:100}}));
   const flow=assessmentWorkflow(store);
   await flow.save(task,'student-1','coach-1',{expectedRevision:0,entries:inputs});
   const first=await flow.publish(task,'student-1','coach-1',1);
   await flow.save(task,'student-1','coach-1',{expectedRevision:1,reason:'更正',entries:[{...inputs[0]!,value:{kind:'score_0_100',score:60}}]});
   const corrected=await flow.publish(task,'student-1','coach-1',2);
   const raw=await confirmedStudentResults(store,clubId,'student-1',new Set([task.teamId]));
   expect(raw.find(r=>r.testItemId===inputs[0]!.testItemId)?.value).toEqual({kind:'score_0_100',score:60});
   const records=store.getStudentMetrics(clubId,'student-1');
   const changedMetric=store.listAssessmentTestItems(clubId).find(i=>i.id===inputs[0]!.testItemId)!.metricId;
   const old=records.find(r=>r.assessmentId===first.assessmentId && r.metricId===changedMetric)!;
   const replacement=records.find(r=>r.assessmentId===corrected.assessmentId && r.metricId===old.metricId)!;
   const growth=await app.inject({method:'GET',url:'/clubs/'+clubId+'/app-clients/app-client-cq-talent-wechat-main/parent/students/student-1/growth-summary',headers:{'x-user-id':'user-parent-1'}});
   expect(growth.statusCode).toBe(200);
   const latest=growth.json().latest.find((r:any)=>r.metricId===old.metricId).record;
   writeFileSync('/tmp/cq-audit-projection-result.json',JSON.stringify({oldId:old.id,oldValue:old.value,newId:replacement.id,newValue:replacement.value,latestId:latest.id,latestValue:latest.value,latestAssessment:latest.assessmentId,expectedAssessment:corrected.assessmentId,oldAssessment:first.assessmentId},null,2));
   expect(old.value).toEqual({kind:'score_0_100',score:100});
   expect(replacement.value).toEqual({kind:'score_0_100',score:60});
   expect(latest.value.score).toBe(100);
   expect(latest.assessmentId).toBe(first.assessmentId);
  }finally{await app.close();p.database.close();vi.unstubAllEnvs();}
 },30000);
});
